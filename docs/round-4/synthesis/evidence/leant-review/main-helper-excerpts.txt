2439:   :: St -> BehavioralRun -> Bool -> String -> IO (Either String JValue)
2440: runBehavioralCommand st active synthesisEnvironment code = do
2441:   before <- behavioralRequestSeconds st active
2442:   case before of
2443:     Nothing -> pure (Left "the synthesis deadline has no time for another check")
2444:     Just _ -> do
2445:       backendOr <- ensureBackend st
2446:       case backendOr of
2447:         Left failure -> pure (Left failure)
2448:         Right backend -> do
2449:           environmentOr <- if synthesisEnvironment
2450:             then fmap (fmap Just) (ensureSynthEnv st)
2451:             else Right . rsEnv <$> readIORef st
2452:           case environmentOr of
2453:             Left failure -> pure (Left failure)
2454:             Right environment -> do
2455:               seconds <- behavioralRequestSeconds st active
2456:               case seconds of
2457:                 Nothing -> pure (Left "the synthesis deadline has no time for another check")
2458:                 Just limit -> do
2459:                   result <- request backend (Just limit) (commandPayload environment code)
2460:                   case result of
2461:                     Right response -> pure (Right response)
2462:                     Left RequestTimeout -> do
2463:                       backendDied st
2464:                       pure (Left $ "decision timed out after " ++ show limit
2465:                         ++ "s; the backend was retired for session replay")
2466:                     Left (ServerClosed stderrText) -> do
2467:                       backendDied st
2468:                       pure (Left $ "the Lean server died during the check: " ++ trim stderrText)
2469:                     Left (BadResponse failure) -> pure (Left $ "bad response: " ++ failure)
