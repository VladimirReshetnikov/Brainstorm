5145: -- | Find and display a useful next tactic without changing the user's proof
5146: -- stack. The backend proof-state protocol is persistent, so speculative
5147: -- children do not affect the state to which the next user tactic is applied.
5148: -- Candidates that merely make progress are remembered (up to three) while
5149: -- the search keeps looking for one that closes the goal; if none does, a
5150: -- second phase chains the quick finishers onto each remembered candidate
5151: -- (`t <;> finisher`) so the suggestion can still be a complete verified
5152: -- proof. The annotation on the shown suggestion says which kind it is.
5153: suggestTactic :: St -> IO ()
5154: suggestTactic st = do
5155:   state <- readIORef st
5156:   case rsProve state of
5157:     Just pv | (ps, goals@(g : _), _) : _ <- pvStack pv ->
5158:       case lookup ps (pvSuggestions pv) of
5159:         Just cached -> forM_ cached (emitSuggestion st)
5160:         Nothing ->
5161:           probe ps (length goals) (chainExtras g) (goalCandidates g) []
5162:     _ -> pure ()
5163:  where
5164:   -- Chain finishers beyond the quick certain ones: `simp_all` can use the
5165:   -- case hypotheses an `induction` or `cases` step introduces, and when
5166:   -- the target mentions definitions, `simp_all [f]` unfolds them -- the
5167:   -- move that closes `induction l <;> simp_all [myLen]` proofs. Probed
5168:   -- after `exact?` so an informative found term still wins over a
5169:   -- sledgehammer when both close the goal.
5170:   chainExtras g =
5171:     "simp_all" : ["simp_all [" ++ d ++ "]" | d <- goalDefNames g]
5172:   -- Run one heartbeat-bounded candidate against the immutable proof state.
5173:   -- The bound keeps proactive help responsive even when a project has a very
5174:   -- large premise database. A heartbeat exhaustion is an ordinary tactic
5175:   -- error, unlike the REPL's wall-clock timeout, so it does not kill the
5176:   -- backend or invalidate the user's proof state. Returns Nothing when the
5177:   -- backend died (the emergency exit has already run), Just Nothing when
5178:   -- Lean rejected the candidate.
5179:   probeOnce :: Integer -> String -> IO (Maybe (Maybe JValue))
5180:   probeOnce ps tactic = do
5181:     let bounded = "set_option maxHeartbeats "
5182:           ++ show suggestionHeartbeatLimit ++ " in " ++ tactic
5183:     result <- runTactic st ps bounded
5184:     case result of
5185:       Left err -> Nothing <$ proveEmergencyExit st
5186:         ("the backend failed while suggesting a tactic: " ++ err)
5187:       Right v
5188:         | Nothing <- respFatal v
5189:         , null [() | (severity, _) <- respMessages v, severity == "error"]
5190:         , Just _ <- respProofState v -> pure (Just (Just v))
5191:         | otherwise -> pure (Just Nothing)
5192: 
5193:   probe ps nGoals extras [] helds = chainPhase ps nGoals extras helds
5194:   probe ps nGoals extras ("intro" : rest) helds
5195:     | any (("intro " `isPrefixOf`) . fst) helds =
5196:         probe ps nGoals extras rest helds  -- the named intro variant held
5197:   probe ps nGoals extras (tactic : rest) helds = do
5198:     outcome <- probeOnce ps tactic
5199:     case outcome of
5200:       Nothing -> pure ()
5201:       Just Nothing -> probe ps nGoals extras rest helds
5202:       Just (Just v) -> do
5203:         let text = fromMaybe tactic (parseTryThis (respMessages v))
5204:             remaining = length (respGoals v)
5205:             -- library-search tactics can "close" the probe state with
5206:             -- metavariables and say so in a Remaining-subgoals comment;
5207:             -- never present those as closing the goal
5208:             partial = any (("-- Remaining subgoals:" `isPrefixOf`) . trim)
5209:               (lines text)
5210:         if remaining < nGoals && not partial
5211:           then conclude ps (Just (annotateSuggestion "closes the goal" text))
5212:           else probe ps nGoals extras rest $ if length helds >= 3 then helds
5213:             else helds ++ [(text, if remaining > nGoals
5214:                    then Just ("splits into "
5215:                      ++ show (remaining - nGoals + 1) ++ " goals")
5216:                    else Nothing)]
5217: 
5218:   -- No single candidate closed the goal: try to discharge each remembered
5219:   -- candidate's residual subgoals in one more step. `<;>` (rather than `;`)
5220:   -- makes the finisher run on every subgoal the candidate produces, so
5221:   -- acceptance means the chain is a complete proof of the current goal.
5222:   -- `exact?` is only chained onto the first candidate to bound the cost of
5223:   -- the expensive library searches; the `simp_all` extras come after it so
5224:   -- an informative found term beats a sledgehammer. A candidate whose text
5225:   -- spans lines is never chained, since composing it would not yield a
5226:   -- tactic the user could type back. If no chain closes either, the first
5227:   -- remembered candidate is shown with its own annotation.
5228:   chainPhase ps _ _ [] = conclude ps Nothing
5229:   chainPhase ps nGoals extras helds@(first : _) =
5230:     goHeld (zip (True : repeat False) helds)
5231:    where
5232:     fallback = conclude ps (Just (annotateHeld first))
5233:     goHeld [] = fallback
5234:     goHeld ((isFirst, (text, _)) : more)
5235:       | '\n' `elem` text = goHeld more
5236:       | otherwise = goChain text
5237:           (suggestionFinishers ++ ["exact?" | isFirst] ++ extras) more
5238:     goChain _ [] more = goHeld more
5239:     goChain text (fin : rest) more = do
5240:       outcome <- probeOnce ps (text ++ " <;> " ++ fin)
5241:       case outcome of
5242:         Nothing -> pure ()
5243:         Just (Just v)
5244:           | length (respGoals v) < nGoals
5245:           , Just chained <- renderChain text fin v ->
5246:               conclude ps (Just (annotateSuggestion "closes the goal" chained))
5247:           -- a simp_all link that was accepted without closing may have
5248:           -- rewritten the residual goals into omega's arithmetic reach
5249:           -- (`induction n <;> simp_all [f] <;> omega`); one extension probe
5250:           | "simp_all" `isPrefixOf` fin -> do

5254:                 Nothing -> pure ()

5256:                   conclude ps $ Just $ annotateSuggestion "closes the goal"
5257:                     (text ++ " <;> " ++ fin')

5259:         Just _ -> goChain text rest more
5260:     -- The chained text must be something the user can type back verbatim.
5261:     -- A literal finisher composes trivially; `exact?` has to splice in the
5262:     -- term it found, which is only sound when the chain ran it against
5263:     -- exactly one subgoal (one Try-this message) and the found tactic fits
5264:     -- on one line (a Remaining-subgoals comment never does).
5265:     renderChain text "exact?" v = case
5266:       [ d | (sev, d) <- respMessages v, sev == "info"
5267:           , "Try this:" `isPrefixOf` dropWhile isSpace d ] of
5268:       [only] | Just found <- parseTryThis [("info", only)]
5269:              , '\n' `notElem` found ->
5270:         Just (text ++ " <;> " ++ found)
5271:       _ -> Nothing
5272:     renderChain text fin _ = Just (text ++ " <;> " ++ fin)
5273: 
5274:   annotateHeld (text, note) = maybe text (`annotateSuggestion` text) note
5275:

5277:     cache ps suggestion
5278:     forM_ suggestion (emitSuggestion st)
